// two-sum — O(n) with a hash map
func twoSum(_ nums: [Int], _ target: Int) -> [Int] {
var seen: [Int: Int] = [:]
for (i, n) in nums.enumerated() {
let need = target - n
if let j = seen[need] {
return [j, i]
}
seen[n] = i
}
return []
}
// walk-through
// nums = [2, 7, 11, 15], target = 9
let result = twoSum([2, 7, 11, 15], 9)
print("indices: \(result)")
// complexity: O(n) time, O(n) space
